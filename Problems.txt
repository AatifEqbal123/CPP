std::string checker(int marks) {
    std::string status;
    std::string grade;
    std::string res;
    
    if (marks >= 80) {
            status = "Great, eligible for scholarship ";
            (marks >= 90) ? grade="A" : grade="B";         
    }
    
   else if (marks >= 70 and marks <= 79){
       grade="C";
   }
   else if(marks >= 60 and marks <= 69){
       grade="D";
   }
   else {
       grade = "F";
   }  
   res = (marks>=60) ? "Passed, Sucessfully" : "Sorry, try again";
   
   std::string final = status + "\n" + "grade : " + grade + "\n" + "result : " + res;
   return final;
}
